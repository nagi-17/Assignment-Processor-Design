// Instruction Encoders (Defined before use to prevent compiler errors)

inline uint32_t encode_r(uint8_t op, uint8_t rd, uint8_t rs1, uint8_t rs2) {
    return ((uint32_t)(op & 0x3F) << 26) | 
           ((uint32_t)(rd & 0x0F) << 21) | 
           ((uint32_t)(rs1 & 0x0F) << 17) | 
           ((uint32_t)(rs2 & 0x0F) << 13);
}

inline uint32_t encode_i(uint8_t op, uint8_t rd, uint8_t rs1, uint16_t imm) {
    return ((uint32_t)(op & 0x3F) << 26) | 
           (1u << 25) | 
           ((uint32_t)(rd & 0x0F) << 21) | 
           ((uint32_t)(rs1 & 0x0F) << 17) | 
           ((uint32_t)(imm & 0xFFFF) << 1);
}
